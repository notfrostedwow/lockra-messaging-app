-- Simple fix for users table RLS policy to prevent recursion
BEGIN;
  -- Disable RLS temporarily to fix the issue
  ALTER TABLE public.users DISABLE ROW LEVEL SECURITY;
  
  -- Drop all existing policies on the users table
  DROP POLICY IF EXISTS "Enable read access for all users" ON public.users;
  DROP POLICY IF EXISTS "Enable insert for authenticated users only" ON public.users;
  DROP POLICY IF EXISTS "Enable update for users based on email" ON public.users;
  DROP POLICY IF EXISTS "Enable delete for users based on email" ON public.users;
  DROP POLICY IF EXISTS "Users can read own data and admins can read all" ON public.users;
  DROP POLICY IF EXISTS "Users can update own data and admins can update all" ON public.users;
  DROP POLICY IF EXISTS "Authenticated users can insert own data" ON public.users;
  DROP POLICY IF EXISTS "Only admins can delete users" ON public.users;
  
  -- Create simple policies
  
  -- Select policy
  CREATE POLICY "Select Policy" ON public.users
    FOR SELECT USING (true);
  
  -- Update policy
  CREATE POLICY "Update Policy" ON public.users
    FOR UPDATE USING (true);
  
  -- Insert policy
  CREATE POLICY "Insert Policy" ON public.users
    FOR INSERT WITH CHECK (true);
  
  -- Delete policy
  CREATE POLICY "Delete Policy" ON public.users
    FOR DELETE USING (true);
  
  -- Re-enable RLS
  ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;
COMMIT;
