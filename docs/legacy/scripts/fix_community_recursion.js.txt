require('dotenv').config();
const { createClient } = require('@supabase/supabase-js');

// Initialize Supabase client with service role key for admin access
const supabaseUrl = process.env.SUPABASE_URL;
const supabaseServiceKey = process.env.SUPABASE_SERVICE_KEY;

if (!supabaseUrl || !supabaseServiceKey) {
  console.error('Error: SUPABASE_URL and SUPABASE_SERVICE_KEY must be set in the .env file');
  process.exit(1);
}

const supabase = createClient(supabaseUrl, supabaseServiceKey);

async function runSql(sql, description) {
  console.log(`Running SQL: ${description}...`);
  const { data, error } = await supabase.rpc('exec_sql', { sql });
  
  if (error) {
    console.error(`Error running SQL (${description}):`, error);
    return false;
  }
  
  console.log(`Success: ${description}`);
  return true;
}

async function fixCommunityPolicies() {
  console.log('Starting community policies fix...');
  
  try {
    // Fix community_members policies
    await runSql(`
      -- Drop existing problematic policies
      DROP POLICY IF EXISTS "Members can see who else is in their communities" ON community_members;
      DROP POLICY IF EXISTS "Anyone can view public community members" ON community_members;
      DROP POLICY IF EXISTS "Members can see private community members" ON community_members;
    `, 'Drop community_members SELECT policies');
    
    // Create simpler community_members policies
    await runSql(`
      -- Simple policy for everyone to select (will be filtered by RLS)
      CREATE POLICY "Allow select on community_members" 
        ON community_members FOR SELECT 
        TO authenticated
        USING (TRUE);
    `, 'Create simplified community_members SELECT policy');

    // Fix user's ability to join communities
    await runSql(`
      DROP POLICY IF EXISTS "Users can join communities" ON community_members;
      CREATE POLICY "Users can join communities" 
        ON community_members FOR INSERT 
        TO authenticated
        WITH CHECK (user_id = auth.uid());
    `, 'Fix join communities policy');

    // Fix role management
    await runSql(`
      DROP POLICY IF EXISTS "Owners and admins can update member roles" ON community_members;
      CREATE POLICY "Owners and admins can update member roles" 
        ON community_members FOR UPDATE 
        TO authenticated
        USING (
          EXISTS (
            SELECT 1 FROM community_members cm
            WHERE cm.community_id = community_members.community_id
            AND cm.user_id = auth.uid()
            AND cm.role IN ('owner', 'admin')
          )
        );
    `, 'Fix role management policy');

    // Fix leave communities
    await runSql(`
      DROP POLICY IF EXISTS "Members can leave communities" ON community_members;
      CREATE POLICY "Members can leave communities" 
        ON community_members FOR DELETE 
        TO authenticated
        USING (user_id = auth.uid());
    `, 'Fix leave communities policy');

    // Fix community policies
    await runSql(`
      DROP POLICY IF EXISTS "Members can view their private communities" ON communities;
      DROP POLICY IF EXISTS "Anyone can view public communities" ON communities;
    `, 'Drop communities SELECT policies');

    await runSql(`
      -- Simple policy for public communities
      CREATE POLICY "Anyone can view communities" 
        ON communities FOR SELECT 
        USING (TRUE);
    `, 'Create simplified communities SELECT policy');

    // Fix other community policies
    await runSql(`
      DROP POLICY IF EXISTS "Authenticated users can create communities" ON communities;
      CREATE POLICY "Authenticated users can create communities" 
        ON communities FOR INSERT 
        TO authenticated
        WITH CHECK (auth.uid() IS NOT NULL);
        
      DROP POLICY IF EXISTS "Owners can update their communities" ON communities;
      CREATE POLICY "Owners can update their communities" 
        ON communities FOR UPDATE 
        TO authenticated
        USING (owner_id = auth.uid());
        
      DROP POLICY IF EXISTS "Owners can delete their communities" ON communities;
      CREATE POLICY "Owners can delete their communities" 
        ON communities FOR DELETE 
        TO authenticated
        USING (owner_id = auth.uid());
    `, 'Fix other communities policies');

    // Fix community_messages policies
    await runSql(`
      DROP POLICY IF EXISTS "Members can see messages in their communities" ON community_messages;
      DROP POLICY IF EXISTS "Members can send messages to their communities" ON community_messages;
      DROP POLICY IF EXISTS "Users can delete their own messages" ON community_messages;
    `, 'Drop community_messages policies');

    await runSql(`
      -- Members can see messages in their communities
      CREATE POLICY "Members can see messages in their communities" 
        ON community_messages FOR SELECT 
        TO authenticated
        USING (
          EXISTS (
            SELECT 1 FROM community_members cm
            WHERE cm.community_id = community_messages.community_id 
            AND cm.user_id = auth.uid()
          )
        );
        
      -- Members can send messages to their communities
      CREATE POLICY "Members can send messages to their communities" 
        ON community_messages FOR INSERT 
        TO authenticated
        WITH CHECK (
          user_id = auth.uid() AND
          EXISTS (
            SELECT 1 FROM community_members cm
            WHERE cm.community_id = community_messages.community_id 
            AND cm.user_id = auth.uid()
          )
        );
        
      -- Users can only delete their own messages
      CREATE POLICY "Users can delete their own messages" 
        ON community_messages FOR DELETE 
        TO authenticated
        USING (user_id = auth.uid());
    `, 'Create community_messages policies');

    // Enable realtime for community_messages
    await runSql(`
      -- Enable realtime for community_messages
      ALTER PUBLICATION supabase_realtime ADD TABLE IF NOT EXISTS community_messages;
    `, 'Enable realtime for community_messages');

    console.log('Successfully fixed community policies!');
  } catch (error) {
    console.error('Error fixing community policies:', error);
  }
}

// Run the fix
fixCommunityPolicies();
