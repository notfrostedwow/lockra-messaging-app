-- Migration to fix messages table RLS policies for Clerk authentication
-- This migration updates the RLS policies to work with Clerk auth instead of Supabase Auth

-- Drop existing policies for messages table
DROP POLICY IF EXISTS "Users can insert messages they send" ON public.messages;
DROP POLICY IF EXISTS "Users can insert messages they send with Clerk auth" ON public.messages;
DROP POLICY IF EXISTS "Users can read messages they sent or received" ON public.messages;
DROP POLICY IF EXISTS "Users can read messages they sent or received with Clerk auth" ON public.messages;
DROP POLICY IF EXISTS "Users can update messages they sent" ON public.messages;
DROP POLICY IF EXISTS "Users can update messages they sent with Clerk auth" ON public.messages;
DROP POLICY IF EXISTS "Users can delete their own messages" ON public.messages;
DROP POLICY IF EXISTS "Users can delete their own messages with Clerk auth" ON public.messages;

-- Create new policies that work with Clerk authentication
-- For the insert policy, we need to ensure the sender_id matches a user where the clerk_id matches auth.uid()

-- Insert policy
CREATE POLICY "Users can insert messages they send with Clerk auth"
ON public.messages FOR INSERT
WITH CHECK (
  EXISTS (
    SELECT 1 FROM users
    WHERE users.id = sender_id
    AND users.clerk_id = auth.uid()::text
  )
);

-- Read policy
CREATE POLICY "Users can read messages they sent or received with Clerk auth"
ON public.messages FOR SELECT
USING (
  EXISTS (
    SELECT 1 FROM users
    WHERE users.id = sender_id
    AND users.clerk_id = auth.uid()::text
  )
  OR 
  EXISTS (
    SELECT 1 FROM users 
    WHERE users.clerk_id = auth.uid()::text 
    AND users.username = receiver_username
  )
);

-- Update policy
CREATE POLICY "Users can update messages they sent with Clerk auth"
ON public.messages FOR UPDATE
USING (
  EXISTS (
    SELECT 1 FROM users
    WHERE users.id = sender_id
    AND users.clerk_id = auth.uid()::text
  )
);

-- Delete policy
CREATE POLICY "Users can delete their own messages with Clerk auth"
ON public.messages FOR DELETE
USING (
  EXISTS (
    SELECT 1 FROM users
    WHERE users.id = sender_id
    AND users.clerk_id = auth.uid()::text
  )
);

-- Also update the file_attachments policies
DROP POLICY IF EXISTS "Users can insert their attachments" ON public.file_attachments;
DROP POLICY IF EXISTS "Users can insert their attachments with Clerk auth" ON public.file_attachments;

CREATE POLICY "Users can insert their attachments with Clerk auth"
ON public.file_attachments FOR INSERT
WITH CHECK (
  EXISTS (
    SELECT 1 FROM messages
    JOIN users ON users.id = messages.sender_id
    WHERE messages.id = file_attachments.message_id
    AND users.clerk_id = auth.uid()::text
  )
);

DROP POLICY IF EXISTS "Users can delete their attachments" ON public.file_attachments;
DROP POLICY IF EXISTS "Users can delete their attachments with Clerk auth" ON public.file_attachments;

CREATE POLICY "Users can delete their attachments with Clerk auth"
ON public.file_attachments FOR DELETE
USING (
  EXISTS (
    SELECT 1 FROM messages
    JOIN users ON users.id = messages.sender_id
    WHERE messages.id = file_attachments.message_id
    AND users.clerk_id = auth.uid()::text
  )
);

DROP POLICY IF EXISTS "Users can read their attachments" ON public.file_attachments;
DROP POLICY IF EXISTS "Users can read their attachments with Clerk auth" ON public.file_attachments;

CREATE POLICY "Users can read their attachments with Clerk auth"
ON public.file_attachments FOR SELECT
USING (
  EXISTS (
    SELECT 1 FROM messages
    JOIN users sender ON sender.id = messages.sender_id
    WHERE messages.id = file_attachments.message_id
    AND (
      sender.clerk_id = auth.uid()::text
      OR
      EXISTS (
        SELECT 1 FROM users receiver
        WHERE receiver.clerk_id = auth.uid()::text 
        AND receiver.username = messages.receiver_username
      )
    )
  )
);

-- Add comment explaining what we did
COMMENT ON TABLE public.messages IS 'Messages between users with RLS policies updated for Clerk authentication';
