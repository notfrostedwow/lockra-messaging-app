-- Create enum types for community types and member roles
CREATE TYPE community_type AS ENUM ('public', 'private');
CREATE TYPE member_role AS ENUM ('owner', 'admin', 'member');

-- Create communities table
CREATE TABLE IF NOT EXISTS communities (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name TEXT NOT NULL,
  type community_type NOT NULL,
  invite_code TEXT UNIQUE,
  owner_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now() NOT NULL,
  
  -- Ensure private communities have an invite code
  CONSTRAINT valid_community_config CHECK (
    (type = 'public') OR 
    (type = 'private' AND invite_code IS NOT NULL)
  )
);

-- Create community_members table
CREATE TABLE IF NOT EXISTS community_members (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  community_id UUID NOT NULL REFERENCES communities(id) ON DELETE CASCADE,
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  role member_role NOT NULL,
  joined_at TIMESTAMP WITH TIME ZONE DEFAULT now() NOT NULL,
  
  -- Ensure a user can only be a member of a community once
  CONSTRAINT unique_community_member UNIQUE (community_id, user_id)
);

-- Create community_messages table
CREATE TABLE IF NOT EXISTS community_messages (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  community_id UUID NOT NULL REFERENCES communities(id) ON DELETE CASCADE,
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  content TEXT NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now() NOT NULL
);

-- Enable Row Level Security
ALTER TABLE communities ENABLE ROW LEVEL SECURITY;
ALTER TABLE community_members ENABLE ROW LEVEL SECURITY;
ALTER TABLE community_messages ENABLE ROW LEVEL SECURITY;

-- Create policies for communities table
-- Anyone can view public communities
CREATE POLICY "Anyone can view public communities" 
  ON communities FOR SELECT 
  USING (type = 'public');

-- Members can view their private communities
CREATE POLICY "Members can view their private communities" 
  ON communities FOR SELECT 
  USING (
    type = 'private' AND 
    EXISTS (
      SELECT 1 FROM community_members 
      WHERE community_id = communities.id AND user_id = auth.uid()
    )
  );

-- Only authenticated users can create communities
CREATE POLICY "Authenticated users can create communities" 
  ON communities FOR INSERT 
  WITH CHECK (auth.uid() IS NOT NULL);

-- Only community owners can update their communities
CREATE POLICY "Owners can update their communities" 
  ON communities FOR UPDATE 
  USING (owner_id = auth.uid());

-- Only community owners can delete their communities
CREATE POLICY "Owners can delete their communities" 
  ON communities FOR DELETE 
  USING (owner_id = auth.uid());

-- Create policies for community_members table
-- Members can see who else is in their communities
CREATE POLICY "Members can see who else is in their communities" 
  ON community_members FOR SELECT 
  USING (
    EXISTS (
      SELECT 1 FROM community_members AS cm
      WHERE cm.community_id = community_members.community_id 
      AND cm.user_id = auth.uid()
    )
  );

-- Users can join communities (will be controlled by application logic)
CREATE POLICY "Users can join communities" 
  ON community_members FOR INSERT 
  WITH CHECK (user_id = auth.uid());

-- Only owners and admins can update member roles
CREATE POLICY "Owners and admins can update member roles" 
  ON community_members FOR UPDATE 
  USING (
    EXISTS (
      SELECT 1 FROM community_members AS cm
      WHERE cm.community_id = community_members.community_id 
      AND cm.user_id = auth.uid()
      AND cm.role IN ('owner', 'admin')
    )
  );

-- Members can leave communities (delete themselves)
CREATE POLICY "Members can leave communities" 
  ON community_members FOR DELETE 
  USING (user_id = auth.uid());

-- Create policies for community_messages table
-- Members can see messages in their communities
CREATE POLICY "Members can see messages in their communities" 
  ON community_messages FOR SELECT 
  USING (
    EXISTS (
      SELECT 1 FROM community_members
      WHERE community_id = community_messages.community_id 
      AND user_id = auth.uid()
    )
  );

-- Members can send messages to their communities
CREATE POLICY "Members can send messages to their communities" 
  ON community_messages FOR INSERT 
  WITH CHECK (
    user_id = auth.uid() AND
    EXISTS (
      SELECT 1 FROM community_members
      WHERE community_id = community_messages.community_id 
      AND user_id = auth.uid()
    )
  );

-- Users can only delete their own messages
CREATE POLICY "Users can delete their own messages" 
  ON community_messages FOR DELETE 
  USING (user_id = auth.uid());

-- Function to generate a random 8-character invite code
CREATE OR REPLACE FUNCTION generate_invite_code()
RETURNS TEXT AS $$
DECLARE
  chars TEXT := 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
  result TEXT := '';
  i INT;
BEGIN
  FOR i IN 1..8 LOOP
    result := result || substr(chars, floor(random() * length(chars) + 1)::integer, 1);
  END LOOP;
  RETURN result;
END;
$$ LANGUAGE plpgsql VOLATILE;

-- Trigger to automatically generate invite code for private communities
CREATE OR REPLACE FUNCTION set_invite_code_for_private_community()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.type = 'private' AND (NEW.invite_code IS NULL OR NEW.invite_code = '') THEN
    -- Keep generating until we get a unique code
    LOOP
      NEW.invite_code := generate_invite_code();
      EXIT WHEN NOT EXISTS (SELECT 1 FROM communities WHERE invite_code = NEW.invite_code);
    END LOOP;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER ensure_private_community_has_invite_code
BEFORE INSERT OR UPDATE ON communities
FOR EACH ROW
EXECUTE FUNCTION set_invite_code_for_private_community();

-- Trigger to automatically add the creator as owner when a community is created
CREATE OR REPLACE FUNCTION add_owner_to_community()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO community_members (community_id, user_id, role)
  VALUES (NEW.id, NEW.owner_id, 'owner');
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER add_creator_as_owner
AFTER INSERT ON communities
FOR EACH ROW
EXECUTE FUNCTION add_owner_to_community();
