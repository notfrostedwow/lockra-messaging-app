-- Drop the existing function
DROP FUNCTION IF EXISTS public.revoke_user_sessions;

-- Recreate the function without the problematic update
CREATE OR REPLACE FUNCTION public.revoke_user_sessions(user_id_param UUID)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
  -- Delete all refresh tokens for the user
  DELETE FROM auth.refresh_tokens
  WHERE auth.refresh_tokens.user_id = user_id_param;
  
  -- Update the user's last_sign_in_at to invalidate existing sessions
  UPDATE auth.users
  SET last_sign_in_at = '1970-01-01 00:00:00+00'::timestamptz
  WHERE id = user_id_param;
  
  -- No longer updating the banned flag here to avoid recursion
  -- The banned flag is updated directly in the application code
END;
$$;

-- Create a fallback function that only handles token revocation
